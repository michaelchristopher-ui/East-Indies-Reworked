#	FLR_loyal_lords out of FLR_total_lords sets the lords modifier:
#	none loyal = -10% stability, -0.15 political power; all loyal = +10%, +0.15
FLR_update_lords = {
	clamp_variable = {
		var = FLR_loyal_lords
		min = 0
		max = FLR_total_lords
	}
	set_temp_variable = { lords_share = FLR_loyal_lords }
	divide_temp_variable = { lords_share = FLR_total_lords }

	set_variable = { FLR_lords_stability = lords_share }
	multiply_variable = { FLR_lords_stability = 0.2 }
	subtract_from_variable = { FLR_lords_stability = 0.1 }

	set_variable = { FLR_lords_pp = lords_share }
	multiply_variable = { FLR_lords_pp = 0.3 }
	subtract_from_variable = { FLR_lords_pp = 0.15 }

	force_update_dynamic_modifier = yes
}
