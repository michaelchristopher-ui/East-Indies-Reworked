country_event = { #Larantuka 1 - End of the regency
	id = flr_swapraja.1
	title = flr_swapraja.1.t
	desc = flr_swapraja.1.d

	picture = GFX_report_event_generic_sign_treaty

	is_triggered_only = no
	fire_only_once = yes

	trigger = {
		original_tag = LRT
		date > 1938.1.1
		has_idea = LRT_raja_under_regency
	}

	mean_time_to_happen = {
		days = 1
	}

	option = { #Long live Raja Lorenzo III!
		name = flr_swapraja.1.a
		remove_ideas = LRT_raja_under_regency
		recruit_character = LRT_lorenzo_iii
		promote_character = LRT_lorenzo_iii
		set_country_flag = LRT_raja_kedua
		add_stability = 0.05
	}
}

country_event = { #Flores 10 - Rajas of unreleased kingdoms serve the Indies
	id = flr_swapraja.10
	hidden = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				country_exists = NTT
			}
			FLR_move_servus_to_ntt = yes
			FLR_move_thomas_to_ntt = yes
			FLR_move_bagoeng_to_ntt = yes
		}
		else = {
			FLR_move_servus_to_ins = yes
			FLR_move_thomas_to_ins = yes
			FLR_move_bagoeng_to_ins = yes
		}
	}

	option = {
		name = flr_swapraja.10.a
	}
}

#	The same moments for the holder of the state when the kingdom is not a country

country_event = { #Larantuka 1938 - Lorenzo III takes the throne
	id = flr_swapraja.20
	title = flr_swapraja.20.t
	desc = flr_swapraja.20.d

	picture = GFX_report_event_generic_sign_treaty

	is_triggered_only = no
	fire_only_once = yes

	trigger = {
		OR = {
			original_tag = INS
			original_tag = EIS
			original_tag = NTT
		}
		owns_state = 1087
		NOT = { country_exists = LRT }
		date > 1938.1.1
	}

	mean_time_to_happen = {
		days = 1
	}

	option = {
		name = flr_swapraja.20.a
		# the nearer the holder is to the raja, the more the event matters
		if = {
			limit = { original_tag = NTT }
			add_political_power = 15
		}
		else_if = {
			limit = { original_tag = EIS }
			add_political_power = 10
		}
		else = {
			add_political_power = 5
		}
	}
}
