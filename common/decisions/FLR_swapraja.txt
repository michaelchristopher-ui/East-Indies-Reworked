	FLR_recognise_larantuka = {
		icon = generic_independence
		cost = 50
		fire_only_once = yes

		visible = {
			NOT = { country_exists = LRT }
		}

		available = {
			has_war = no
			owns_state = 1087
			controls_state = 1087
		}

		complete_effect = {
			release = LRT
			# an independent Lesser Sunda is the federal level whenever it exists
			if = {
				limit = {
					NOT = { original_tag = EIS }
					country_exists = EIS
					EIS = { is_subject = no }
				}
				EIS = {
					set_autonomy = {
						target = LRT
						autonomous_state = autonomy_federal_member
					}
				}
			}
			else_if = {
				limit = { is_subject = no }
				set_autonomy = {
					target = LRT
					autonomous_state = autonomy_federal_member
				}
			}
			# subjects can't hold subjects, so the member goes to the top of the chain
			else_if = {
				limit = { OVERLORD = { is_subject = no } }
				OVERLORD = {
					set_autonomy = {
						target = LRT
						autonomous_state = autonomy_federal_member
					}
				}
			}
			else = {
				OVERLORD = {
					OVERLORD = {
						set_autonomy = {
							target = LRT
							autonomous_state = autonomy_federal_member
						}
					}
				}
			}
		}

		ai_will_do = {
			base = 0
		}
	}
