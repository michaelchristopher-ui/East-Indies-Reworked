#	Move a kingdom's raja to the Indies when the kingdom is not released

FLR_move_servus_to_ins = {
	every_possible_country = {
		limit = {
			has_character = LRT_servus
		}
		random_character = {
			limit = {
				is_character = LRT_servus
			}
			set_nationality = INS
		}
	}
}
